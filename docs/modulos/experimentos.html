Planejamento e Análise de Experimentos

Delineamentos experimentais (DIC, DBC, DQL, Fatorial) e testes de médias

Delineamentos Básicos

Delineamento Inteiramente Casualizado (DIC)

res_dic <- analise_dic(y = dados$produtividade, trat = dados$adubo, teste_medias = "tukey")
grafico_dic(res_dic)

Delineamento em Blocos Casualizados (DBC)

res_dbc <- analise_dbc(y = dados$produtividade, trat = dados$adubo, bloco = dados$bloco)
grafico_dbc(res_dbc)

Delineamento Quadrado Latino (DQL)

res_dql <- analise_dql(y = dados$prod, trat = dados$trat, linha = dados$lin, coluna = dados$col)
grafico_dql(res_dql)

Esquemas Fatoriais e Parcelas

Fatorial Duplo e Adicional

# Fatorial Duplo (Fator A x Fator B)
res_fat <- analise_fatorial_duplo(y = dados$prod, fator_a = dados$irrigacao, fator_b = dados$adubo)
grafico_fatorial(res_fat)

# Fatorial com Testemunha Adicional
res_fat_ad <- analise_fatorial_adicional(y = dados$prod, fator_a = dados$fA, fator_b = dados$fB, testemunha = dados$test)

Parcelas Subdivididas (Split-Plot)

res_split <- analise_parcela_subdividida(y = dados$prod, parcela = dados$p, subparcela = dados$sp, bloco = dados$b)
